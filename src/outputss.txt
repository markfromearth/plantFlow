import React, { createContext, useContext, useState, ReactNode } from 'react';
import { Equipment, Booking, EquipmentStatus, BookingStatus } from './types';
import { INITIAL_EQUIPMENT, INITIAL_BOOKINGS } from './mockData';
import { addEquipment as dbAddEquipment, updateEquipmentStatus as dbUpdateEquipmentStatus, listEquipment as dbListEquipment } from './equipmentService';
import { createBooking as dbCreateBooking, updateBookingStatus as dbUpdateBookingStatus, listBookings as dbListBookings } from './bookingService';

interface StoreState {
  equipment: Equipment[];
  bookings: Booking[];
  highVisMode: boolean;
  isLoadingEquipment: boolean;
  isLoadingBookings: boolean;
  errorEquipment: string | null;
  errorBookings: string | null;
}

interface StoreContextType extends StoreState {
  addBooking: (booking: Booking) => Promise<void>;
  updateBookingStatus: (id: string, status: BookingStatus) => void;
  updateEquipmentStatus: (id: string, status: EquipmentStatus) => void;
  addEquipment: (equipment: Equipment) => void;
  startHire: (id: string) => void;
  endHire: (id: string) => void;
  toggleHighVisMode: () => void;
  init: () => Promise<void>;

const StoreContext = createContext<StoreContextType | undefined>(undefined);

export const StoreProvider: React.FC<{ children: ReactNode }> = ({ children }) => {
  const [equipment, setEquipment] = useState<Equipment[]>(INITIAL_EQUIPMENT);
  const [bookings, setBookings] = useState<Booking[]>(INITIAL_BOOKINGS);
  const [highVisMode, setHighVisMode] = useState(false);
  const [isLoadingEquipment, setIsLoadingEquipment] = useState(false);
  const [isLoadingBookings, setIsLoadingBookings] = useState(false);
  const [errorEquipment, setErrorEquipment] = useState<string | null>(null);
  const [errorBookings, setErrorBookings] = useState<string | null>(null);

  const init = async () => {
    setIsLoadingEquipment(true);
    setIsLoadingBookings(true);
    setErrorEquipment(null);
    setErrorBookings(null);

    try {
      const equips = await dbListEquipment();
      if (equips.length > 0) {
        setEquipment(equips);
      }
    } catch (e: any) {
      console.warn("Firebase list equipment failed, falling back to local data.", e.message);
      // We don't set errorEquipment so the UI doesn't show an error state
    } finally {
      setIsLoadingEquipment(false);
    }

    try {
      const dbBookings = await dbListBookings();
      if (dbBookings.length > 0) {
        setBookings(dbBookings);
      }
    } catch (e: any) {
      console.warn("Firebase list bookings failed, falling back to local data.", e.message);
      // We don't set errorBookings so the UI doesn't show an error state
    } finally {
      setIsLoadingBookings(false);
    }
  };

  const toggleHighVisMode = () => setHighVisMode(prev => !prev);


  const addBooking = async (booking: Booking) => {
    setIsLoadingBookings(true);
    setErrorBookings(null);
    try {
      const payload: Omit<Booking, 'booking_id' | 'timestamp_created'> = {
        machine_id: booking.machine_id,
        hirer_name: booking.hirer_name,
        hirer_phone: booking.hirer_phone,
        hirer_email: booking.hirer_email,
        site_address: booking.site_address,
        start_date: booking.start_date,
        end_date: booking.end_date,
        delivery_window: booking.delivery_window,
        status: booking.status,
        qr_start_code: booking.qr_start_code,
        qr_end_code: booking.qr_end_code,
        totalCost: booking.totalCost,
        deliveryNeeded: booking.deliveryNeeded
      };
      
      const newBooking = await dbCreateBooking(payload);
      setBookings(prev => [...prev, newBooking]);
    } catch (e: any) {
      console.error("Firebase add booking failed:", e.message);
      setErrorBookings(e.message || "Failed to add booking");
      throw e;
    } finally {
      setIsLoadingBookings(false);
    }
  };

  const updateBookingStatus = async (id: string, status: BookingStatus) => {
    setIsLoadingBookings(true);
    setErrorBookings(null);
    try {
      await dbUpdateBookingStatus(id, status);
    } catch (e: any) {
      console.warn("Firebase update booking status failed, updating local state only.", e.message);
    } finally {
      setBookings(prev => prev.map(b => b.booking_id === id ? { ...b, status } : b));
      setIsLoadingBookings(false);
    }
  };

  const updateEquipmentStatus = async (id: string, status: EquipmentStatus) => {
    setIsLoadingEquipment(true);
    setErrorEquipment(null);
    try {
      await dbUpdateEquipmentStatus(id, status);
    } catch (e: any) {
      console.warn("Firebase update equipment status failed, updating local state only.", e.message);
    } finally {
      setEquipment(prev => prev.map(e => e.machine_id === id ? { ...e, availability_status: status } : e));
      setIsLoadingEquipment(false);
    }
  };

  const addEquipment = async (equipment: Equipment) => {
    setIsLoadingEquipment(true);
    setErrorEquipment(null);
    try {
      const payload: Omit<Equipment, 'machine_id'> = {
        name: equipment.name,
        category: equipment.category,
        job_fit: equipment.job_fit,
        access_width_mm: equipment.access_width_mm,
        day_rate: equipment.day_rate,
        delivery_rate: equipment.delivery_rate,
        availability_status: equipment.availability_status,
        compliance_expiry: equipment.compliance_expiry,
        image_url: equipment.image_url,
        notes: equipment.notes
      };
      
      const newEquip = await dbAddEquipment(payload);
      setEquipment(prev => [...prev, newEquip]);
    } catch (e: any) {
      console.warn("Firebase add equipment failed, falling back to local state.", e.message);
      const newEquip: Equipment = {
        ...equipment,
        machine_id: `mock-equip-${Date.now()}`,
      };
      setEquipment(prev => [...prev, newEquip]);
    } finally {
      setIsLoadingEquipment(false);
    }
  };

  const startHire = async (id: string) => {
    const booking = bookings.find(b => b.booking_id === id);
    if (!booking) return;
    
    setIsLoadingBookings(true);
    setErrorBookings(null);
    try {
      await dbUpdateBookingStatus(id, 'active');
      await dbUpdateEquipmentStatus(booking.machine_id, 'hired');
    } catch (e: any) {
      console.warn("Firebase start hire failed, updating local state only.", e.message);
    } finally {
      setBookings(prev => prev.map(b => b.booking_id === id ? { ...b, status: 'active' } : b));
      setEquipment(prev => prev.map(e => e.machine_id === booking.machine_id ? { ...e, availability_status: 'hired' } : e));
      setIsLoadingBookings(false);
    }
  };

  const endHire = async (id: string) => {
    const booking = bookings.find(b => b.booking_id === id);
    if (!booking) return;
    
    setIsLoadingBookings(true);
    setErrorBookings(null);
    try {
      await dbUpdateBookingStatus(id, 'completed');
      await dbUpdateEquipmentStatus(booking.machine_id, 'available');
    } catch (e: any) {
      console.warn("Firebase end hire failed, updating local state only.", e.message);
    } finally {
      setBookings(prev => prev.map(b => b.booking_id === id ? { ...b, status: 'completed' } : b));
      setEquipment(prev => prev.map(e => e.machine_id === booking.machine_id ? { ...e, availability_status: 'available' } : e));
      setIsLoadingBookings(false);
    }
  };

  return (
    <StoreContext.Provider value={{
      equipment,
      bookings,
      highVisMode,
      isLoadingEquipment,
      isLoadingBookings,
      errorEquipment,
      errorBookings,
      addBooking,
      updateBookingStatus,
      updateEquipmentStatus,
      addEquipment,
      startHire,
      endHire,
      toggleHighVisMode,
      init
    }}>
      {children}
    </StoreContext.Provider>
  );
};

export const useStore = () => {
  const context = useContext(StoreContext);
  if (context === undefined) {
    throw new Error('useStore must be used within a StoreProvider');
  }
  return context;
};

import {
  signInWithPopup,
  GoogleAuthProvider,
  onAuthStateChanged,
  User,
  signOut
} from 'firebase/auth';

import { auth, googleProvider } from './firebase';

let isSigningIn = false;
let cachedAccessToken: string | null = null;

// Google-only auth listener
export const initAuth = (
  onAuthSuccess?: (user: User, token: string) => void,
  onAuthFailure?: () => void
) => {
  return onAuthStateChanged(auth, async (user: User | null) => {
    if (user) {
      // If we already have a cached token, use it
      if (cachedAccessToken && onAuthSuccess) {
        onAuthSuccess(user, cachedAccessToken);
        return;
      }

      // Try to refresh the token from the current user
      const token = await user.getIdToken();
      cachedAccessToken = token;

      if (onAuthSuccess) onAuthSuccess(user, token);
    } else {
      // No user signed in — Google login required
      cachedAccessToken = null;
      if (onAuthFailure) onAuthFailure();
    }
  });
};

export const googleSignIn = async (): Promise<{ user: User; accessToken: string } | null> => {
  try {
    isSigningIn = true;

    const result = await signInWithPopup(auth, googleProvider);
    const credential = GoogleAuthProvider.credentialFromResult(result);

    if (!credential?.accessToken) {
      throw new Error('Failed to get access token from Firebase Auth');
    }

    cachedAccessToken = credential.accessToken;

    return { user: result.user, accessToken: cachedAccessToken };
  } catch (err: unknown) {
    if (err instanceof Error) {
      console.error('Sign in error:', err.message);
    } else {
      console.error('Sign in error:', String(err));
    }
    return null;
  } finally {
    isSigningIn = false;
  }
};

export const getAccessToken = async (): Promise<string | null> => {
  return cachedAccessToken;
};

export const logout = async () => {
  await signOut(auth);
  cachedAccessToken = null;
};

export async function uploadEquipmentImage(file: File): Promise<string> {
  const storage = getStorage();
  const uuid = typeof crypto !== 'undefined' && crypto.randomUUID 
    ? crypto.randomUUID() 
    : Math.random().toString(36).substring(2, 15);
    
  const storageRef = ref(storage, `equipment/${uuid}`);
  await uploadBytes(storageRef, file);
  const downloadUrl = await getDownloadURL(storageRef);
  return downloadUrl;
}

import { collection, doc, getDocs, getDoc, addDoc, updateDoc } from 'firebase/firestore';
import { db } from './firebase';
import { Equipment, EquipmentStatus } from './types';

// Your Firestore rules require tenant‑scoped collections.
// This MUST match: /tenants/{tenantId}/inventory
const TENANT_ID = 'plantflow-os';

export async function listEquipment(): Promise<Equipment[]> {
  const equipmentCol = collection(db, `tenants/${TENANT_ID}/inventory`);
  const equipmentSnapshot = await getDocs(equipmentCol);

  return equipmentSnapshot.docs.map(doc => ({
    machine_id: doc.id,
    ...doc.data()
  }) as Equipment);
}

export async function getEquipment(id: string): Promise<Equipment | null> {
  const equipmentDoc = doc(db, `tenants/${TENANT_ID}/inventory/${id}`);
  const equipmentSnapshot = await getDoc(equipmentDoc);

  if (!equipmentSnapshot.exists()) return null;

  return {
    machine_id: equipmentSnapshot.id,
    ...equipmentSnapshot.data()
  } as Equipment;
}

export async function addEquipment(payload: Omit<Equipment, 'machine_id'>): Promise<Equipment> {
  const equipmentCol = collection(db, `tenants/${TENANT_ID}/inventory`);
  const docRef = await addDoc(equipmentCol, payload);

  return {
    machine_id: docRef.id,
    ...payload
  } as Equipment;
}

export async function updateEquipmentStatus(id: string, availability_status: EquipmentStatus): Promise<void> {
  const equipmentDoc = doc(db, `tenants/${TENANT_ID}/inventory/${id}`);
  await updateDoc(equipmentDoc, { availability_status });
}

import React, { ErrorInfo, ReactNode } from 'react';
import { AlertTriangle, Home } from 'lucide-react';
import { Link } from 'react-router-dom';

interface Props {
  children?: ReactNode;
  fallback?: ReactNode;
}

interface State {
  hasError: boolean;
  error: Error | null;
}

export class ErrorBoundary extends React.Component<Props, State> {
  public state: State = {
    hasError: false,
    error: null,
  };

  public static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  public componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error('Uncaught error:', error, errorInfo);
  }

  public render() {
    if (this.state.hasError) {
      if (this.props.fallback) {
        return this.props.fallback;
      }

      return (
        <div className="min-h-screen bg-slate-50 flex items-center justify-center p-4">
          <div className="max-w-md w-full bg-white rounded-3xl p-8 shadow-sm border border-slate-200 text-center">
            <div className="bg-red-50 w-16 h-16 rounded-full flex items-center justify-center mx-auto mb-6">
              <AlertTriangle className="w-8 h-8 text-red-500" />
            </div>
            <h1 className="text-2xl font-bold text-slate-900 mb-2">Something went wrong</h1>
            <p className="text-slate-600 mb-6">
              {this.state.error?.message || "An unexpected error occurred in this component."}
            </p>
            <div className="flex gap-3 justify-center">
              <button
                onClick={() => this.setState({ hasError: false, error: null })}
                className="px-6 py-3 bg-slate-100 text-slate-700 font-bold rounded-xl hover:bg-slate-200 transition-colors"
              >
                Try Again
              </button>
              <Link
                to="/"
                onClick={() => this.setState({ hasError: false, error: null })}
                className="px-6 py-3 bg-indigo-600 text-white font-bold rounded-xl hover:bg-indigo-700 transition-colors inline-flex items-center gap-2"
              >

import { collection, doc, getDocs, getDoc, addDoc, updateDoc, query, where, orderBy } from 'firebase/firestore';
import { db } from './firebase';
import { Booking, BookingStatus } from './types';

// Your Firestore rules require tenant‑scoped collections.
// This MUST match: /tenants/{tenantId}/bookings
const TENANT_ID = 'plantflow-os';

export async function listBookings(status?: BookingStatus): Promise<Booking[]> {
  const bookingsCol = collection(db, `tenants/${TENANT_ID}/bookings`);

  let q = query(bookingsCol, orderBy('timestamp_created', 'desc'));

  if (status) {
    q = query(
      bookingsCol,
      where('status', '==', status),
      orderBy('timestamp_created', 'desc')
    );
  }

  const bookingsSnapshot = await getDocs(q);

  return bookingsSnapshot.docs.map(doc => ({
    booking_id: doc.id,
    ...doc.data()
  }) as Booking);
}

export async function getBooking(id: string): Promise<Booking | null> {
  const bookingDoc = doc(db, `tenants/${TENANT_ID}/bookings/${id}`);
  const bookingSnapshot = await getDoc(bookingDoc);

  if (!bookingSnapshot.exists()) return null;

  return {
    booking_id: bookingSnapshot.id,
    ...bookingSnapshot.data()
  } as Booking;
}

export async function createBooking(payload: Omit<Booking, 'booking_id' | 'timestamp_created'>): Promise<Booking> {
  const bookingsCol = collection(db, `tenants/${TENANT_ID}/bookings`);

  const timestamp_created = new Date().toISOString();
  const bookingData = { ...payload, timestamp_created };

  const docRef = await addDoc(bookingsCol, bookingData);

  return {
    booking_id: docRef.id,
    ...bookingData
  } as Booking;
}

export async function updateBookingStatus(id: string, status: BookingStatus): Promise<void> {
  const bookingDoc = doc(db, `tenants/${TENANT_ID}/bookings/${id}`);
  await updateDoc(bookingDoc, { status });
}

import { useState, useCallback, useEffect } from 'react';
import { googleSignIn, initAuth, logout } from '../authService';
import { User } from 'firebase/auth';

export function useOAuthToken() {
  const [token, setToken] = useState<string | null>(null);
  const [user, setUser] = useState<User | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [isLoading, setIsLoading] = useState(false);
  const [needsAuth, setNeedsAuth] = useState(true);

  useEffect(() => {
    const unsubscribe = initAuth(
      (user, token) => {
        setUser(user);
        setToken(token);
        setNeedsAuth(false);
      },
      () => {
        setUser(null);
        setToken(null);
        setNeedsAuth(true);
      }
    );
    return () => unsubscribe();
  }, []);

  const requestToken = useCallback(async () => {
    setIsLoading(true);
    setError(null);
    try {
      const result = await googleSignIn();
      if (result) {
        setToken(result.accessToken);
        setUser(result.user);
        setNeedsAuth(false);
      }
    } catch (err: any) {
      setError(err.message || 'Login failed');
    } finally {
      setIsLoading(false);
    }
  }, []);

  const handleLogout = useCallback(async () => {
    await logout();
    setToken(null);
    setUser(null);
    setNeedsAuth(true);
  }, []);

  return { token, user, needsAuth, error, isLoading, requestToken, logout: handleLogout };
}
